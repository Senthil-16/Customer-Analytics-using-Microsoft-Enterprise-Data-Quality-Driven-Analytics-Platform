Data_Cleaning_Silver_Layer :
# Silver layer: clean and validate the raw Bronze tables.
# Source tables: bronze_customers, bronze_orders  (raw CSV data loaded into the Lakehouse)
from pyspark.sql.functions import to_date

# ---------- Customers ----------
df_customers = spark.sql("SELECT * FROM bronze_customers")
df_customers.show()

# 1. Remove duplicate customers (customer_id is the business key)
df_no_duplicates = df_customers.dropDuplicates(["customer_id"])

# 2. Drop rows with no email (cannot identify or contact the customer)
df_valid_email = df_no_duplicates.filter("email IS NOT NULL")

# 3. Fill missing city with "Unknown" so the customer stays in city-wise reports
df_city_fixed = df_valid_email.fillna({"city": "Unknown"})

# 4. Standardise the signup date format
df_clean_customers = df_city_fixed.withColumn(
    "signup_date",
    to_date("signup_date", "yyyy-MM-dd")
)

df_clean_customers.show()
df_clean_customers.write.mode("overwrite").format("delta").saveAsTable("silver_customers")

# ---------- Orders ----------
df_orders = spark.sql("SELECT * FROM bronze_orders")
df_orders.show()

# Keep only valid, successful orders: positive amount and completed payment
df_orders_valid = df_orders.filter("order_amount > 0 AND payment_status = 'Completed'")
df_orders_valid.write.mode("overwrite").format("delta").saveAsTable("silver_orders")


Gold_Analytics_Layer :
# Gold layer: business-ready customer summary.
# FIX: the first version calculated total_orders, total_spent and avg_order_value from
# silver_customers only (a count of customer rows), so the money columns were not real values.
# Now silver_orders is joined and the order amounts are summed and averaged.
# A LEFT JOIN keeps customers who have no completed orders (shown with zeros).

gold_customer_summary = spark.sql("""
SELECT
    c.customer_id,
    c.customer_name,
    c.city,
    COUNT(o.order_amount)                      AS total_orders,
    COALESCE(SUM(o.order_amount), 0)           AS total_spent,
    COALESCE(AVG(o.order_amount), 0)           AS avg_order_value
FROM silver_customers c
LEFT JOIN silver_orders o
    ON c.customer_id = o.customer_id
GROUP BY
    c.customer_id,
    c.customer_name,
    c.city
""")

gold_customer_summary.show()

gold_customer_summary.write \
    .mode("overwrite") \
    .format("delta") \
    .saveAsTable("gold_customer_summary")

# Quick checks
spark.sql("SELECT * FROM gold_customer_summary").show()
spark.sql("SHOW TABLES").show()

# Sanity check: total revenue in Gold should equal the sum of completed orders in Silver
spark.sql("SELECT SUM(total_spent) AS gold_revenue FROM gold_customer_summary").show()
spark.sql("SELECT SUM(order_amount) AS silver_revenue FROM silver_orders").show()
